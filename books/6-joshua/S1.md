(function(){
  var root=document.getElementById('aib'); if(!root) return;
  // With no web address (the Word build) the book stays open, every page in order.
  if(!/^https?:$/.test(location.protocol)) return;
  var pgs=[].slice.call(root.querySelectorAll('.aib-pg')), n=pgs.length, i=0;
  var prev=root.querySelector('.aib-prev'), next=root.querySelector('.aib-next'), ct=root.querySelector('.aib-ct');
  var m=location.hash.match(/\/s1\/(\d+)/); if(m) i=Math.min(n,Math.max(1,+m[1]))-1;
  root.classList.add('flip');
  function show(k){ i=Math.max(0,Math.min(n-1,k));
    pgs.forEach(function(p,j){p.classList.toggle('on',j===i)});
    ct.textContent=(i+1)+' / '+n; prev.disabled=i===0; next.disabled=i===n-1; }
  prev.onclick=function(){show(i-1)}; next.onclick=function(){show(i+1)};
  pgs.forEach(function(p){ var f=p.querySelector('.aib-frame'), x0=null;
    f.addEventListener('click',function(e){ var r=f.getBoundingClientRect(); show(e.clientX<r.left+r.width/2?i-1:i+1); });
    f.addEventListener('touchstart',function(e){x0=e.touches[0].clientX},{passive:true});
    f.addEventListener('touchend',function(e){ if(x0===null) return; var dx=e.changedTouches[0].clientX-x0; x0=null;
      if(Math.abs(dx)>40){ e.preventDefault(); show(dx<0?i+1:i-1); } }); });
  function key(e){ if(!document.body.contains(root)){document.removeEventListener('keydown',key);return}
    if(e.key==='ArrowRight') show(i+1); else if(e.key==='ArrowLeft') show(i-1); }
  document.addEventListener('keydown',key);
  show(i);
})();
